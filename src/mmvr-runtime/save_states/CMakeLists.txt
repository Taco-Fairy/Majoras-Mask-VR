cmake_minimum_required(VERSION 3.26)
project(MMVRSaveStates LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mmvr-state-archive STATIC Archive.cpp LiveGraph.cpp RestorePlan.cpp Ownership.cpp)
target_include_directories(mmvr-state-archive PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(mmvr-state-archive PRIVATE /W4 /EHsc)
else()
 target_compile_options(mmvr-state-archive PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mmvr-state-tests ArchiveTests.cpp)
target_link_libraries(mmvr-state-tests PRIVATE mmvr-state-archive)
enable_testing()
add_test(NAME exact-state-archive COMMAND mmvr-state-tests "${CMAKE_CURRENT_BINARY_DIR}/state-test-data")

# Explicit save/load preparation benchmark; never runs during gameplay.
add_executable(mmvr-state-benchmark RestoreBenchmark.cpp)
target_link_libraries(mmvr-state-benchmark PRIVATE mmvr-state-archive)

add_executable(mmvr-native-literal-tests NativeLiteralTests.cpp)
target_compile_definitions(mmvr-native-literal-tests PRIVATE MMVR_ENABLE MMVR_STATE_NATIVE_BACKEND)
target_include_directories(mmvr-native-literal-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_test(NAME native-state-literals COMMAND mmvr-native-literal-tests)
